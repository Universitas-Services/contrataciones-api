import { Injectable } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';

@Injectable()
export class NormativaPliegoService {
  constructor(private prisma: PrismaService) {}

  async obtenerNormativaParaTemplate(idExpediente: number) {
    // 1. Consultar la relación intermedia cargando ambas tablas maestras
    const relaciones = await this.prisma.tb_expediente_normativa.findMany({
      where: { id_expediente_au_au: idExpediente },
      include: {
        tb_normativa_global: true,
        tb_normativa_ente: true,
      },
    });

    // 2. Clasificar y mapear las normativas seleccionadas en strings limpios
    const normativasGlobalesList: string[] = [];
    const normativasEnteList: string[] = [];

    relaciones.forEach((rel) => {
      if (rel.tb_normativa_global && rel.tb_normativa_global.ind_activo) {
        normativasGlobalesList.push(`- ${rel.tb_normativa_global.texto_normativa_completo.trim()}`);
      }
      if (rel.tb_normativa_ente) {
        normativasEnteList.push(`- ${rel.tb_normativa_ente.texto_normativa_completo.trim()}`);
      }
    });

    // 3. Devolver los bloques de texto unificados con salto de carro (
)
    return {
      normativa_general: normativasGlobalesList.join('
'),
      normativa_legal_au_au: normativasEnteList.join('
'),
    };
  }
}
